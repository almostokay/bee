        Google Sheet steps
        1. Make a new Google Sheet. Name it anything.
        2. Extensions → Apps Script. Delete what’s there and paste the script below.
        3. Deploy → New deployment → Web app. Execute as: Me. Who has access: Anyone. Deploy and copy the URL.
        4. Paste the URL above. Add one expense. A row appears with the header: Date | Amount | Category | Description | Payment Method | Month | Year | ID.
        var HEAD = ['Date','Amount','Category','Description','Payment Method','Month','Year','ID'];
function doPost(e) {
  var lock = LockService.getScriptLock(); lock.waitLock(10000);
  try {
    var p = e.parameter, ss = SpreadsheetApp.getActiveSpreadsheet();
    var s = ss.getSheetByName('Expenses') || ss.insertSheet('Expenses');
    if (s.getLastRow() == 0) { s.appendRow(HEAD); s.setFrozenRows(1); }
    s.getRange(1,1,1,8).setValues([HEAD]).setFontWeight('bold');
    var act = p.action || 'add';
    if (act == 'add') { s.appendRow(rowOf(p)); }
    else {
      var r = findRow(s, p);
      if (act == 'delete' && r) s.deleteRow(r);
      if (act == 'update') { if (r) s.getRange(r,1,1,8).setValues([rowOf(p)]); else s.appendRow(rowOf(p)); }
    }
  } finally { lock.releaseLock(); }
  return ContentService.createTextOutput('ok');
}
function rowOf(p) {
  var d = new Date(p.date);
  return [p.date, Number(p.amount), p.category, p.description, p.payment,
    d.toLocaleString('en-US',{month:'long'}), d.getFullYear(), p.id];
}
function findRow(s, p) {
  var v = s.getDataRange().getValues(), i;
  for (i = v.length - 1; i > 0; i--) if (String(v[i][7]) === String(p.id)) return i + 1;
  for (i = v.length - 1; i > 0; i--) {  // rows saved before IDs existed
    if (!v[i][7] && day(v[i][0]) === p.o_date && Number(v[i][1]) === Number(p.o_amount) &&
        v[i][2] === p.o_category && String(v[i][3]) === p.o_description && v[i][4] === p.o_payment) return i + 1;
  }
  return 0;
}
function day(d) { return d instanceof Date ? Utilities.formatDate(d, Session.getScriptTimeZone(), 'yyyy-MM-dd') : String(d); }
